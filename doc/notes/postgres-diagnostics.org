* idle connections
** query all idle transactions
SELECT
    pid,
    usename,
    datname,
    client_addr,
    now() - state_change AS idle_for
FROM pg_stat_activity
WHERE state = 'idle'
ORDER BY idle_for DESC;
** query transactions idle for a certain length of time
SELECT pid, datname, usename, application_name, now() - state_change AS idle_for
FROM pg_stat_activity
WHERE state = 'idle'
  AND now() - state_change > interval '10 minutes'
ORDER BY idle_for desc;
** kill transactions idle after a certain length of time
SELECT pg_terminate_backend(pid)
FROM pg_stat_activity
WHERE datname in ('...DB_NAME...')
  AND state = 'idle'
  AND now() - state_change > interval '10 minutes';
* list all connections by age
SELECT pid,
       usename,
       datname,
       client_addr,
       state,
       now() - state_change AS idle_for,
       now() - backend_start AS age
FROM pg_stat_activity
WHERE backend_type = 'client backend'
  AND pid <> pg_backend_pid()
ORDER BY backend_start ASC
LIMIT 50;
